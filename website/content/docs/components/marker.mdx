## Usage

```tsx showLineNumbers
import { Marker, MarkerContent, MarkerIcon } from "vic-ui/ui"
```

```tsx showLineNumbers

  <MarkerIcon>
    <CheckIcon />
  </MarkerIcon>
  <MarkerContent>Explored 4 files</MarkerContent>
</Marker>
```
