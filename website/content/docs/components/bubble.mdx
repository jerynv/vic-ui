<ComponentPreview name="bubble" />

## Usage

```tsx showLineNumbers
import { Bubble, BubbleContent, BubbleReactions } from "vic-ui/ui"
```

```tsx showLineNumbers
<Bubble>
  <BubbleContent>
    I checked the registry output and removed the stale route.
  </BubbleContent>
  <BubbleReactions>
    <span>👍</span>
  </BubbleReactions>
</Bubble>
```
